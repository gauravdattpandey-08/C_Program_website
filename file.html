Question 1 : Create a file and copy it into another file

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    void main()
    {
        FILE *f1, *f2;
        char ch;

        clrscr();

        f1 = fopen("first.txt", "w");
        fprintf(f1, "Hello, this is my first file.");
        fclose(f1);

        f1 = fopen("first.txt", "r");
        f2 = fopen("copy.txt", "w");

        while ((ch = fgetc(f1)) != -1)
        {
             fputc(ch, f2);
        }

         fclose(f1);
        fclose(f2);

        printf("File copied successfully.");
        delay(3000);
    }

Output:

File copied successfully.

Question 2 : Create two files and concatenate them into a third file

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    void main()
    {
        FILE *f1, *f2, *f3;
        char ch;

        clrscr();

         f1 = fopen("first.txt", "w");
        fprintf(f1, "This is the first file.\n");
        fclose(f1);

        f2 = fopen("second.txt", "w");
        fprintf(f2, "This is the second file.\n");
        fclose(f2);

        f1 = fopen("first.txt", "r");
        f2 = fopen("second.txt", "r");
        f3 = fopen("third.txt", "w");

        while ((ch = fgetc(f1)) != -1)
        {
            fputc(ch, f3);
        }

        while ((ch = fgetc(f2)) != -1)
        {
            fputc(ch, f3);
        }

        fclose(f1);
        fclose(f2);
        fclose(f3);

        printf("Files concatenated successfully.");
        delay(3000);
    }

Output:

Files concatenated successfully

Question 3 : Create a file and count the number of characters

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    void main()
    {
        FILE *fp;
        char ch;
        int count = 0;

        clrscr();

         fp = fopen("data.txt", "w");
        fprintf(fp, "Hello World");
        fclose(fp);

        fp = fopen("data.txt", "r");

        while ((ch = fgetc(fp)) != -1)
        {
            count++;
        }

        fclose(fp);

        printf("Number of characters = %d", count);
        delay(3000);
    }

Output:

                    data entered: hello
                    Number of characters = 5
