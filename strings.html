Question 1 : Enter a string from user and concatenate the string ?

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    #include <string.h>
    void main()
    {
        char str1[20], str2[20];

        clrscr();

        printf("Enter first string: ");
        scanf(" %s", str1);

        printf("Enter second string: ");
        scanf(" %s", str2);

        strcat(str1, str2);

        printf("Combined string: %s", str1);
        delay(3000);
    }

Output:

Gaurav and Pandey -> Gaurav Pandey

Question 2 : Enter a string from user and compare those string ?

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    #include <string.h>
    void main()
    {
        char str1[20], str2[20];

        clrscr();

        printf("Enter first string: ");
        scanf("%s", str1);

        printf("Enter second string: ");
        scanf("%s", str2);

        if(strcmp(str1, str2) == 0)
            printf("Strings are equal");
        else
            printf("Strings are not equal");

        delay(3000);
    }

Output:

Gaurav and Gaurav → Strings are equal

Question 3 : Enter a string from user and copy that string into another string ?

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    #include <string.h>
    void main()
    {
        char str1[20], str2[20];

        clrscr();

        printf("Enter a string: ");
        scanf("%s", str1);

        strcpy(str2, str1);

        printf("Original string: %s\n", str1);
        printf("Copied string: %s", str2);
        delay(3000);
    }

Output:

Hence, the string will be copied.
