Question 1 : Create structure of employee which there is emp.ID, name, designation of 50 employees from user, print details in tabular form ?

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>

    struct employee
    {
        int id;
         char name[30];
        char designation[30];
    };

    void main()
    {
        struct employee e[50];
        int i;
    
        clrscr();

        for(i = 0; i < 50; i++)
            {
                printf("Enter employee ID: ");
                scanf("%d", &e[i].id);

                printf("Enter name: ");
                scanf("%s", e[i].name);

                printf("Enter designation: ");
                scanf("%s", e[i].designation);
            }

        printf("\nID\tName\tDesignation\n");

        for(i = 0; i < 50; i++)
            {
                printf("%d\t%s\t%s\n",
                e[i].id, e[i].name, e[i].designation);
            }
        delay(3000);
    }

Output:

Enter employee ID: 101
Enter name: Gaurav
Enter designation: Manager

Enter employee ID: 102
Enter name: Aman
Enter designation: Clerk...

ID      Name    Designation
101     Rahul   Manager
102     Aman    Clerk ...
