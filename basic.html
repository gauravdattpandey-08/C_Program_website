Question 1 : Write a program to print the message: Today I made my first C Program.

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    void main()
    {
        clrscr();
        printf("Today I made my first C Program.");
        delay(3000);
    }

Output:

Today I made my first C Program.

Question 2 : Write a program to read a character from the keyboard and display it on the screen.

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>

    void main()
    {
        char ch;

        clrscr();

        printf("Enter a character: ");
        scanf("%c", &ch);

        printf("You entered: %c", ch);

        delay(3000);
    }

Output:

Enter a character: A
You entered: A

Question 3 : Write a program to read only 4 characters from the keyboard and display it on the screen in a separate line.

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    void main()
    {
        char a, b, c, d;

        clrscr();

        printf("Enter 4 characters: ");
        scanf("%c%c%c%c", &a, &b, &c, &d);

        printf("\n%c\n%c\n%c\n%c", a, b, c, d);

        delay(3000);
    }

Output:

Enter 4 characters: ABCD
A
B
C
D

Question 6 : Write a program for printing your name on output screen but your name should blink 5 times

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    void main()
    {
        int i;
        clrscr()
        for(i = 1; i <= 5; i++)
        {
        printf("Gaurav");
        delay(500);

        clrscr();
        delay(500);
        }
        delay(3000);
    }

Output:

Gaurav will blink 5 times.

Question 10 : Enter two characters from user and print ASCII value.

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    void main()
    {
        char a, b;

        clrscr();

        printf("Enter first character: ");
        scanf("%c", &a);

        printf("Enter second character: ");
        scanf(" %c", &b);

        printf("ASCII value of %c = %d\n", a, a);
        printf("ASCII value of %c = %d", b, b);

        delay(3000);
    }

Output:

Enter first character: A
Enter second character: B
ASCII value of A = 65
ASCII value of B = 66
