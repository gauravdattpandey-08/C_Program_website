Question 1 : Write a program to show the example of pre-increment ?

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    void main()
    {
        int a = 5;

        clrscr();

        printf("Value before increment = %d\n", a);

        printf("Value after increment = %d", ++a);
        delay(3000);
    }

Output:

Value before increment = 5
Value after increment = 6

Question 5 : Write a program to enter 2 numbers from user add the number and show the answer , again take 2 number from user subtract the number display the answer ,again take 2 number from user multiply the number display the answer and ,again take 2 number from the user divide the number display the answer ,again take 2 number find the remainder and display the answer ?

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    void main()
    {
        int a, b;

        clrscr();

        printf("Enter two numbers for addition: ");
        scanf("%d%d", &a, &b);
        printf("Addition = %d\n", a + b);

        printf("Enter two numbers for subtraction: ");
        scanf("%d%d", &a, &b);
        printf("Subtraction = %d\n", a - b);

        printf("Enter two numbers for multiplication: ");
        scanf("%d%d", &a, &b);
        printf("Multiplication = %d\n", a * b);

        printf("Enter two numbers for division: ");
        scanf("%d%d", &a, &b);

        if(b != 0)
            printf("Division = %d\n", a / b);
        else
            printf("Division is not possible\n");

        printf("Enter two numbers for remainder: ");
        scanf("%d%d", &a, &b);

        if(b != 0)
            printf("Remainder = %d", a % b);
        else
            printf("Remainder is not possible");
        delay(3000);
    }

Output:

Enter two numbers for addition: 10 5
Addition = 15
Enter two numbers for subtraction: 10 5
Subtraction = 5
Enter two numbers for multiplication: 10 5
Multiplication = 50
Enter two numbers for division: 10 5
Division = 2
Enter two numbers for remainder: 10 3
Remainder = 1

Question 11 : Enter marks of students and calculate average of student’s marks ?

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    void main()
    {
        int a, b, c, d, e;
        float average;

        clrscr();

        printf("Enter marks of 5 subjects: ");
        scanf("%d%d%d%d%d", &a, &b, &c, &d, &e);

        average = (a + b + c + d + e) / 5.0;

        printf("Average marks = %f", average)
        delay(3000);
    }

Output:

Enter marks of 5 subjects: 80 70 90 60 100
Average marks = 80.000000

Question 12 : Calculate simple interest = (P x R x T)/ 100

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    void main()
    {
        float p, r, t, si;

        clrscr();

        printf("Enter principal amount: ");
        scanf("%f", &p);

        printf("Enter rate of interest: ");
        scanf("%f", &r);

        printf("Enter time in years: ");
        scanf("%f", &t);

        si = (p * r * t) / 100;

        printf("Simple Interest = %f", si);
        delay(3000);
    }

Output:

Enter principal amount: 1000
Enter rate of interest: 5
Enter time in years: 2
Simple Interest = 100.000000
