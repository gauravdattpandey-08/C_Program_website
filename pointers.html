Question 1 : Q. Write a C program using a pointer with an array. A pointer should point to the array. Through the pointer, we will fill all the locations of the array by asking the values from the user and print the array.

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    void main()
    {
        int x[10];
        int i, *ptr;

         clrscr();

        for(i = 0; i < 10; i++)
        {
            ptr = &x[i];

            printf("Enter number %d here: ", i + 1);
            scanf("%d", ptr);
        }

        for(i = 0; i < 10; i++)
        {
            ptr = &x[i];

            printf("The value at index %d is %d\n", i, *ptr);
        }
        delay(3000);
    }
