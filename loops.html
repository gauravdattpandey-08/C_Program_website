Question 1 : Enter a number from the user and check it is a prime number or not ?

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    void main()
    {
        int n, i, count=0;

        clrscr();

        printf("Enter a number: ");
        scanf("%d", &n);

        for(i=1; i<=n; i++)
        {
            if(n % i == 0)
            {
                count++;
            }
        }

        if(n > 1 && count == 2)
            printf("Prime number");
        else
            printf("Not a prime number");
        delay(3000);
    }

Output:

Enter a number: 7
Prime number

Question 2 : Write a program to print your name 1 lakh times ?

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    void main()
    {
        int i;

        clrscr();

        for(i=1; i<=100000; i++)
        {
            printf("Gaurav\n");
        }
        delay(3000);
    }

Output:

Gaurav is printed 100,000 times.
