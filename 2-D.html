Question 1 : Enter the four number from user and print it into matrix

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    void main()
    {
        int a[2][2], i, j;

        clrscr();

        printf("Enter 4 numbers:\n");

        for(i=0; i<2; i++)
        {
            for(j=0; j<2; j++)
            {
                scanf("%d", &a[i][j]);
            }
        }

        printf("Matrix is:\n");

        for(i=0; i<2; i++)
        {
            for(j=0; j<2; j++)
            {
                printf("%d\t", a[i][j]);
            }

            printf("\n");
        }

        delay(3000);
    }

Output:

TEnter 4 numbers:
1 2 3 4
Matrix is:
1    2
3    4

Question 3 : Enter 2 matrix from user and add them ?

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    void main()
    {
        int a[2][2], b[2][2], c[2][2];
        int i, j;

        clrscr();

        printf("Enter first matrix:\n");

        for(i=0; i<2; i++)
        {
            for(j=0; j<2; j++)
            {
                scanf("%d", &a[i][j]);
            }
        }

        printf("Enter second matrix:\n");

        for(i=0; i<2; i++)
        {
            for(j=0; j<2; j++)
            {
                scanf("%d", &b[i][j]);
            }
        }

        for(i=0; i<2; i++)
        {
            for(j=0; j<2; j++)
            {
                c[i][j] = a[i][j] + b[i][j];
            }
        }

        printf("Addition of matrices:\n");

        for(i=0; i<2; i++)
        {
            for(j=0; j<2; j++)
            {
                printf("%d\t", c[i][j]);
            }

            printf("\n");
        }
        delay(3000);
    }

Output:

Enter first matrix:
1 2
3 4
Enter second matrix:
5 6
7 8
Addition of matrices:
6    8
10   12

Question 5 : Enter 2 matrix from user and multiply them ?

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    void main()
    {
        int a[2][2], b[2][2], c[2][2];
        int i, j, k;

        clrscr();

        printf("Enter first matrix:\n");

        for(i=0; i<2; i++)
        {
            for(j=0; j<2; j++)
            {
                scanf("%d", &a[i][j]);
            }
        }

        printf("Enter second matrix:\n");

        for(i=0; i<2; i++)
        {
            for(j=0; j<2; j++)
            {
                scanf("%d", &b[i][j]);
            }
        }

        for(i=0; i<2; i++)
        {
            for(j=0; j<2; j++)
            {
                c[i][j] = 0;

                for(k=0; k<2; k++)
                {
                    c[i][j] = c[i][j] + a[i][k] * b[k][j];
                }
            }
        }

        printf("Multiplication of matrices:\n");

        for(i=0; i<2; i++)
        {
            for(j=0; j<2; j++)
            {
                printf("%d\t", c[i][j]);
            }

            printf("\n");
        }
        delay(3000);
    }

Output:

Enter first matrix:
1 2
3 4
Enter second matrix:
5 6
7 8
Multiplication of matrices:
19   22
43   50
