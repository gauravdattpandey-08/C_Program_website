Question 1 :Create a union Student with roll number, name and fees. Accept and display each member separately.

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    union Student
    {
        int roll;
        char name[10];
        float fees;
    };

    void main()
    {
        union Student s;

        clrscr();

         printf("Enter roll number: ");
        scanf("%d", &s.roll);
        printf("Roll Number: %d\n", s.roll);

        printf("\nEnter name: ");
        scanf("%9s", s.name);
        printf("Name: %s\n", s.name);

        printf("\nEnter fees: ");
        scanf("%f", &s.fees);
        printf("Fees: %f\n", s.fees);
        delay(3000);
    }

Output:

Roll Number: 12
