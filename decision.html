Question 1 : Write a program to enter the number from the user and check it is positive or negative or 0 (zero) ?

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    void main()
    {
        int n;

        clrscr();

        printf("Enter a number: ");
        scanf("%d", &n);

        if(n > 0)
            printf("Positive number");
        else if(n < 0)
            printf("Negative number");
        else
            printf("Zero");
        delay(3000);
    }

Negative number

    {

        printf("Enter marks: ");

B grade

Question 7 : Enter a number from the user if the number is negative then square the number and if it is positive then increment the number 20 times ?

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    void main()
    {
        int n, i;

        clrscr();

        printf("Enter a number: ");
        scanf("%d", &n);

        if(n < 0)
        {
            n = n * n;
            printf("Square = %d", n);
        }
        else if(n > 0)
        {
            for(i = 1; i <= 20; i++)
                n++;

            printf("Result = %d", n);
        }
        else
            printf("Number is zero");
        delay(3000);
    }

Output:

Enter a number: -4
Square = 16

Question 10 : Write a program to calculate area and perimeter of rectangle ?

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    void main()
    {
        int length, breadth, area, perimeter;

        clrscr();

        printf("Enter length: ");
        scanf("%d", &length);

        printf("Enter breadth: ");
        scanf("%d", &breadth);

        area = length * breadth;
        perimeter = 2 * (length + breadth);

        printf("Area = %d\n", area);
        printf("Perimeter = %d", perimeter);
        delay(3000);
    }

Output:

Enter length: 10
Enter breadth: 5
Area = 50
Perimeter = 30
