Question 1 : Swap numbers using Call by address ?

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    void swap(int *a, int *b)
    {
        int temp;

        temp = *a;
        *a = *b;
        *b = temp;
    }

    void main()
    {
        int a, b;

        clrscr();

        printf("Enter two numbers: ");
        scanf("%d%d", &a, &b);

        printf("Before swapping: a = %d, b = %d\n", a, b);

        swap(&a, &b);

        printf("After swapping: a = %d, b = %d", a, b);

        delay(3000);
    }

Output:

Enter two numbers: 10 20
Before swapping: a = 10, b = 20
After swapping: a = 20, b = 10

Question 2 : Swap numbers using Call by function ?

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    int a, b;

    void swap()
    {
        int temp;

        temp = a;
        a = b;
        b = temp;
    }

    void main()
    {
        clrscr();

        printf("Enter two numbers: ");
        scanf("%d%d", &a, &b);

        printf("Before swapping: a = %d, b = %d\n", a, b);

        swap();

        printf("After swapping: a = %d, b = %d", a, b);

        delay(3000);
    }

Output:

Enter two numbers: 5 9
Before swapping: a = 5, b = 9
After swapping: a = 9, b = 5

Question 3 : Make function for find out Cube a user entered number ?

Solution:

    #include <stdio.h>
    #include <conio.h>
    #include <dos.h>
    int cube(int n)
    {
        return n * n * n;
    }

    void main()
    {
        int n, ans;

        clrscr();

        printf("Enter a number: ");
        scanf("%d", &n);

        ans = cube(n);

        printf("Cube = %d", ans);

        delay(3000);
    }

Output:

Enter a number: 4
Cube = 64
